# Tables

## activity
- uid (PK, VARCHAR(32))
- imglink (VARCHAR(128))
- title (VARCHAR(128))
- descriptions (VARCHAR(128))
- createAt (BIGINT)
- state (SMALLINT, DEFAULT=0)

## activityrecord
- uid (PK, VARCHAR(32))
- imglink (VARCHAR(128))
- title (VARCHAR(128))
- descriptions (VARCHAR(128))
- createAt (BIGINT)
- bindactivity (FK to activity.uid)

## users
- uid (PK, VARCHAR(32))
- loginaccount (VARCHAR(128), UNIQUE)
- loginpassword (VARCHAR(256))
- permission (VARCHAR(45))
- uphone (VARCHAR(45))
- gender (VARCHAR(4))
- uemail (VARCHAR(45))
- uname (VARCHAR(45))
- nickname (VARCHAR(45))
- avatar (VARCHAR(128))

## pet
- uid (PK, VARCHAR(32))
- petname (VARCHAR(128))
- petage (INT)
- pettype (VARCHAR(128))
- petgender (VARCHAR(45))
- petcolor (VARCHAR(45))
- imgurl (VARCHAR(128))
- binduser (FK to users.uid)
- address (VARCHAR(128))
- birthday (VARCHAR(128))

## adoption
- uid (PK, VARCHAR(32))
- bindpet (FK to pet.uid)
- binduser (FK to users.uid)
- createAt (BIGINT)
- state (SMALLINT, DEFAULT=0)
- bindrescue (VARCHAR(32))

## announcement
- uid (PK, VARCHAR(32))
- title (VARCHAR(128))
- detail (VARCHAR(1024))
- times (BIGINT)
- idx (INT, AUTO_INCREMENT, UNIQUE)

## application
- uid (PK, VARCHAR(32))
- binduser (FK to users.uid)
- uname (VARCHAR(128))
- uphone (VARCHAR(128))
- uage (INT)
- gender (VARCHAR(128))
- major (VARCHAR(128))
- income (DOUBLE)
- reason (VARCHAR(1024))
- filelink (VARCHAR(128))
- createAt (BIGINT)
- state (SMALLINT, DEFAULT=0)

## article
- uid (PK, VARCHAR(32))
- title (VARCHAR(256))
- imglink (VARCHAR(128))
- author (VARCHAR(128))
- createAt (BIGINT)
- details (VARCHAR(10240))
- state (SMALLINT, DEFAULT=0)

## disease_records
- uid (PK, VARCHAR(32))
- bindpet (FK to pet.uid)
- disease_name (VARCHAR(128))
- treatment_details (VARCHAR(128))
- recovery_status (VARCHAR(128), DEFAULT=0)
- createAt (BIGINT)

## donate
- uid (PK, VARCHAR(32))
- title (VARCHAR(128))
- imglink (VARCHAR(128))
- maxcount (DOUBLE)
- currentcount (DOUBLE)
- des (VARCHAR(512))
- createAt (BIGINT)
- state (SMALLINT, DEFAULT=0)

## donaterecord
- uid (PK, VARCHAR(32))
- binduser (FK to users.uid)
- binddonate (FK to donate.uid)
- payment (DOUBLE)
- createAt (MEDIUMTEXT)

## rescue
- uid (PK, VARCHAR(32))
- bindpet (FK to pet.uid)
- uage (INT)
- gender (VARCHAR(12))
- major (VARCHAR(128))
- income (DOUBLE)
- state (SMALLINT, DEFAULT=0)
- binduser (FK to users.uid)

## vaccination_records
- uid (PK, VARCHAR(32))
- bindpet (FK to pet.uid)
- vaccination_name (VARCHAR(128))
- createAt (BIGINT)

# Relationships
1. activityrecord.bindactivity --> activity.uid
2. pet.binduser --> users.uid
3. adoption.bindpet --> pet.uid
4. adoption.binduser --> users.uid
5. application.binduser --> users.uid
6. disease_records.bindpet --> pet.uid
7. donaterecord.binduser --> users.uid
8. donaterecord.binddonate --> donate.uid
9. rescue.bindpet --> pet.uid
10. rescue.binduser --> users.uid
11. vaccination_records.bindpet --> pet.uid
